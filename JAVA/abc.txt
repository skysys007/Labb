class Stack {
    private int maxSize;
    private int[] stackArray;
    private int top;

    // Constructor
    public Stack(int size) {
        maxSize = size;
        stackArray = new int[maxSize];
        top = -1;
    }

    // Check if stack is full
    public boolean isFull() {
        return (top == maxSize - 1);
    }

    // Check if stack is empty
    public boolean isEmpty() {
        return (top == -1);
    }

    // Insert element into stack (Push)
    public void push(int value) {
        if (isFull()) {
            System.out.println("Stack is Full. Cannot insert element.");
        } else {
            stackArray[++top] = value;
            System.out.println(value + " inserted into stack.");
        }
    }

    // Remove element from stack (Pop)
    public void pop() {
        if (isEmpty()) {
            System.out.println("Stack is Empty. Cannot remove element.");
        } else {
            System.out.println(stackArray[top] + " removed from stack.");
            top--;
        }
    }

    // Display stack elements
    public void display() {
        if (isEmpty()) {
            System.out.println("Stack is Empty.");
        } else {
            System.out.println("Stack elements are:");
            for (int i = top; i >= 0; i--) {
                System.out.println(stackArray[i]);
            }
        }
    }
}

// Main class to test Stack
public class Main {
    public static void main(String[] args) {
        Stack s = new Stack(3);

        s.push(10);
        s.push(20);
        s.push(30);
        s.push(40);  // Stack full

        s.display();

        s.pop();
        s.pop();

        s.display();
    }
}
